  // ===================== API =====================
  function fetchData() {
    if (!ENDPOINT) { apply(buildFixture()); return Promise.resolve(); }
    return fetch(ENDPOINT, { credentials: 'omit', cache: 'no-store' })
      .then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
      .then(function (j) { apply(j); })
      .catch(function (e) { log('fetch error', e); state.error = 'unavailable'; state.loading = false; render(); });
  }
  function apply(j) {
    state.error = (j && j.error) ? j.error : null;
    state.letters = normalize((j && j.letters) || []);
    state.total = state.letters.reduce(function (n, L) { return n + L.terms.length; }, 0);
    state.loading = false;
    render();
  }
  function normalize(list) {
    var out = [];
    for (var i = 0; i < list.length; i++) {
      var L = list[i]; if (!L || !L.letter || !Array.isArray(L.terms)) continue;
      var terms = [];
      for (var k = 0; k < L.terms.length; k++) {
        var tm = L.terms[k]; if (!tm || !tm.term) continue;
        var html = tm.contentHtml == null ? '' : String(tm.contentHtml);
        terms.push({
          id: String(tm.id || (L.letter + '-' + k)),
          term: String(tm.term),
          contentHtml: html,
          plain: htmlToText(html), // Klartext fuer die Suche (einmal beim Laden)
          links: sanitizeLinks(tm.links)
        });
      }
      if (terms.length) out.push({ letter: String(L.letter).charAt(0).toUpperCase(), terms: terms });
    }
    return out;
  }
  function sanitizeLinks(links) {
    if (!Array.isArray(links)) return [];
    var out = [];
    for (var i = 0; i < links.length; i++) {
      var l = links[i]; if (!l || !l.url || !l.label) continue;
      var url = String(l.url);
      // Defense-in-depth: nur echte interne (/… aber NICHT //) oder http(s)-Links — kein
      // javascript:/data: und kein protocol-relative //evil.com (das wuerde sonst als "intern"
      // durchgehen und same-window ohne noopener off-platform navigieren).
      var internal = url.charAt(0) === '/' && url.charAt(1) !== '/';
      if (!(internal || /^https?:\/\//i.test(url))) { log('link skipped (scheme)', url); continue; }
      out.push({ label: String(l.label), url: url, kind: (l.kind === 'hub' || l.kind === 'course') ? l.kind : 'manual' });
    }
    return out;
  }

  // ===================== RENDER HELPERS =====================
  function el(tag, cls, txt) { var n = document.createElement(tag); if (cls) n.className = cls; if (txt != null) n.textContent = txt; return n; }
  function clickable(node, handler) {
    node.setAttribute('role', 'button'); node.setAttribute('tabindex', '0');
    node.addEventListener('click', handler);
    node.addEventListener('keydown', function (ev) { if (ev.key === 'Enter' || ev.key === ' ') { ev.preventDefault(); handler(ev); } });
    return node;
  }
  // Sanitize-Allowlist fuer contentHtml (server-seitig bereits DOMPurify; hier defensiv nochmal).
  var ALLOWED = { P:1, BR:1, B:1, I:1, EM:1, STRONG:1, U:1, UL:1, OL:1, LI:1, A:1, H3:1, SPAN:1 };
  // contentHtml -> gescrubbtes DocumentFragment (inert geparst: keine Loads/on*-Handler beim Parsen).
  function parseClean(html) {
    var tpl = document.createElement('template');
    tpl.innerHTML = html;
    scrub(tpl.content);
    return tpl.content;
  }
  function htmlToText(html) { var f = parseClean(html); return (f.textContent || '').replace(/\s+/g, ' ').trim(); }
  function renderContent(html, q) {
    var box = el('div', 'tc24-gl-content');
    var frag = parseClean(html);
    while (frag.firstChild) box.appendChild(frag.firstChild);
    if (q) highlightNode(box, q); // Fundstellen im (bereits gescrubbten) Text markieren
    return box;
  }
  function scrub(node) {
    var kids = [].slice.call(node.children);
    for (var i = 0; i < kids.length; i++) {
      var c = kids[i];
      if (!ALLOWED[c.tagName]) { node.removeChild(c); continue; }
      var attrs = [].slice.call(c.attributes);
      for (var a = 0; a < attrs.length; a++) {
        var name = attrs[a].name.toLowerCase(), val = attrs[a].value;
        if (name.indexOf('on') === 0) { c.removeAttribute(attrs[a].name); continue; }
        if (name === 'href') {
          var internalHref = val.charAt(0) === '/' && val.charAt(1) !== '/'; // // => extern, nicht "intern"
          if (!(internalHref || /^https?:\/\//i.test(val) || val.charAt(0) === '#')) { c.removeAttribute('href'); }
          else if (/^https?:\/\//i.test(val)) { c.setAttribute('target', '_blank'); c.setAttribute('rel', 'noopener noreferrer'); }
        } else if (name !== 'class') {
          c.removeAttribute(attrs[a].name);
        }
      }
      scrub(c);
    }
  }
  function renderLinks(links) {
    var wrap = el('div', 'tc24-gl-links');
    links.forEach(function (l) {
      var isInternal = l.url.charAt(0) === '/' && l.url.charAt(1) !== '/';
      var a = el('a', 'tc24-gl-link' + (l.kind === 'manual' ? ' tc24-gl-link-manual' : ''));
      a.href = l.url;
      if (!isInternal) { a.target = '_blank'; a.rel = 'noopener noreferrer'; }
      a.appendChild(el('span', 'tc24-gl-link-ico', KIND_ICON[l.kind] || KIND_ICON.manual));
      a.appendChild(el('span', null, l.label));
      wrap.appendChild(a);
    });
    return wrap;
  }
  // Text mit hervorgehobenen Treffern in ein Ziel-Element schreiben (nur selbst erzeugte Spans/Textnodes).
  function highlightInto(parent, text, q) {
    if (!q) { parent.appendChild(document.createTextNode(text)); return; } // Guard gegen Endlosschleife
    var lower = text.toLowerCase(), from = 0, idx;
    while ((idx = lower.indexOf(q, from)) >= 0) {
      if (idx > from) parent.appendChild(document.createTextNode(text.slice(from, idx)));
      parent.appendChild(el('span', 'tc24-gl-mark', text.slice(idx, idx + q.length)));
      from = idx + q.length;
    }
    if (from < text.length) parent.appendChild(document.createTextNode(text.slice(from)));
  }
  // Textnodes eines Elements rekursiv nach q durchsuchen und Treffer markieren.
  function highlightNode(node, q) {
    var kids = [].slice.call(node.childNodes);
    for (var i = 0; i < kids.length; i++) {
      var n = kids[i];
      if (n.nodeType === 3) {
        var text = n.nodeValue || '';
        if (text.toLowerCase().indexOf(q) >= 0) {
          var frag = document.createDocumentFragment();
          highlightInto(frag, text, q);
          node.replaceChild(frag, n);
        }
      } else if (n.nodeType === 1 && n.className !== 'tc24-gl-mark') {
        highlightNode(n, q);
      }
    }
  }

  // ===================== RENDER (Split: Shell einmal, Body bei jeder Interaktion) =====================

      card.appendChild(buildSearchRow());
    }

    bodyEl = el('div', 'tc24-gl-body');
    card.appendChild(bodyEl);
    root.appendChild(card);
    renderBody();
  }

    clear.addEventListener('click', function () { state.query = ''; input.value = ''; input.focus(); renderBody(); this.style.visibility = 'hidden'; });

    row.appendChild(box);
    return row;
  }

  function renderBody() {

    bodyEl.innerHTML = '';
    if (state.loading) {

    } else if (state.error) {
      bodyEl.appendChild(el('div', 'tc24-gl-state', 'Glossar momentan nicht erreichbar. Bitte später erneut versuchen.'));
    } else if (!state.letters.length) {
      bodyEl.appendChild(el('div', 'tc24-gl-state', 'Noch keine Begriffe vorhanden.'));
    } else if (state.query.trim()) {
      renderSearch(bodyEl, state.query.trim().toLowerCase());
    } else {
      renderBrowse(bodyEl);
    }
  }

  // -------- Browse-Modus: 2-Spalten-Accordion --------
  function renderBrowse(container) {
    var grid = el('div', 'tc24-gl-grid');
    state.letters.forEach(function (L) {
      grid.appendChild(state.openLetters[L.letter] ? renderOpenSection(L) : renderLetterTile(L));
    });
    container.appendChild(grid);
  }
  function renderLetterTile(L) {
    var tile = el('div', 'tc24-gl-letter');
    tile.appendChild(el('span', 'tc24-gl-chev', '▶'));
    tile.appendChild(el('span', 'tc24-gl-badge', L.letter));
    tile.appendChild(el('span', 'tc24-gl-count-inline', L.terms.length + (L.terms.length === 1 ? ' Begriff' : ' Begriffe')));
    clickable(tile, function () { state.openLetters[L.letter] = true; renderBody(); });
    tile.setAttribute('aria-expanded', 'false');
    return tile;
  }
  function renderOpenSection(L) {
    var sec = el('div', 'tc24-gl-section');
    var sh = el('div', 'tc24-gl-section-head');
    sh.appendChild(el('span', 'tc24-gl-chev tc24-gl-open', '▶'));
    sh.appendChild(el('span', 'tc24-gl-badge', L.letter));
    sh.appendChild(el('span', 'tc24-gl-count-inline', L.terms.length + (L.terms.length === 1 ? ' Begriff' : ' Begriffe')));
    clickable(sh, function () { state.openLetters[L.letter] = false; renderBody(); });
    sh.setAttribute('aria-expanded', 'true');
    sec.appendChild(sh);
    L.terms.forEach(function (tm) { sec.appendChild(renderTermRow(L, tm, null)); });
    return sec;
  }
  // Eine Begriff-Reihe (+ Inhalt wenn offen). q != null => Suchmodus: immer offen + Highlight.
  function renderTermRow(L, tm, q) {
    var wrap = document.createDocumentFragment();
    var tkey = L.letter + '|' + tm.id;
    var topen = q ? true : !!state.openTerms[tkey];
    var row = el('div', 'tc24-gl-term');
    row.appendChild(el('span', 'tc24-gl-term-dot'));
    var name = el('span', 'tc24-gl-term-name');
    if (q) highlightInto(name, tm.term, q); else name.textContent = tm.term;
    row.appendChild(name);
    row.appendChild(el('span', 'tc24-gl-term-chev' + (topen ? ' tc24-gl-open' : ''), '▶'));
    if (!q) { clickable(row, function () { state.openTerms[tkey] = !topen; renderBody(); }); row.setAttribute('aria-expanded', topen ? 'true' : 'false'); }
    wrap.appendChild(row);
    if (topen) {
      var content = renderContent(tm.contentHtml, q);
      if (tm.links.length) content.appendChild(renderLinks(tm.links));
      wrap.appendChild(content);
    }
    return wrap;
  }

  // -------- Suchmodus: nur Treffer, aufgeklappt, markiert --------
  function renderSearch(container, q) {
    var any = false, hits = 0;
    state.letters.forEach(function (L) {
      var matched = L.terms.filter(function (tm) {
        return tm.term.toLowerCase().indexOf(q) >= 0 || tm.plain.toLowerCase().indexOf(q) >= 0;
      });
      if (!matched.length) return;
      any = true; hits += matched.length;
      var sec = el('div', 'tc24-gl-section');
      var sh = el('div', 'tc24-gl-section-head tc24-gl-static');
      sh.appendChild(el('span', 'tc24-gl-badge', L.letter));
      sh.appendChild(el('span', 'tc24-gl-count-inline', matched.length + ' Treffer'));
      sec.appendChild(sh);
      matched.forEach(function (tm) { sec.appendChild(renderTermRow(L, tm, q)); });
      container.appendChild(sec);
    });
    if (!any) {
      container.appendChild(el('div', 'tc24-gl-state', 'Keine Treffer für „' + state.query.trim() + '".'));
    }
  }

  // ===================== THEME (an TC24-BMS angelehnt) =====================
  var LIGHT = {
    '--tc24-gl-card-bg': 'rgb(255,255,255)', '--tc24-gl-elev-bg': 'rgb(246,247,248)',
    '--tc24-gl-text': 'rgb(28,57,65)', '--tc24-gl-text-2': 'rgb(55,74,82)', '--tc24-gl-text-3': 'rgb(107,114,120)',
    '--tc24-gl-border': 'rgb(228,230,232)', '--tc24-gl-hover': 'rgba(28,57,65,0.055)', '--tc24-gl-chip-off': 'rgb(237,239,241)',
    '--tc24-gl-mark-bg': 'rgb(254,240,138)', '--tc24-gl-mark-fg': 'rgb(28,57,65)'
  };
  var DARK = {
    '--tc24-gl-card-bg': 'rgb(51,51,51)', '--tc24-gl-elev-bg': 'rgba(255,255,255,0.05)',
    '--tc24-gl-text': 'rgb(245,245,245)', '--tc24-gl-text-2': 'rgba(255,255,255,0.82)', '--tc24-gl-text-3': 'rgba(255,255,255,0.55)',
    '--tc24-gl-border': 'rgba(255,255,255,0.14)', '--tc24-gl-hover': 'rgba(255,255,255,0.07)', '--tc24-gl-chip-off': 'rgba(255,255,255,0.06)',
    '--tc24-gl-mark-bg': 'rgba(250,204,21,0.38)', '--tc24-gl-mark-fg': 'rgb(255,255,255)'
  };
  var lastDark = null;
  // Ganze Ancestor-Kette hochlaufen (LS verschachtelt tief) bis zum ersten ausreichend opaken
  // Hintergrund. Faint Overlays (rgba …,0.04) sind kein Background -> ueberspringen. Komplett
  // transparente Kette rendert auf weissem Canvas -> hell (Fallback false). (CC-25 v0.7.1 Muster)
  function isDark(node) {
    while (node) {
      var bg = getComputedStyle(node).backgroundColor, m = bg && bg.match(/[\d.]+/g);
      if (m && m.length >= 3) {
        var alpha = m.length >= 4 ? Number(m[3]) : 1;
        if (alpha >= 0.5) return (0.299 * m[0] + 0.587 * m[1] + 0.114 * m[2]) / 255 < 0.5;
      }
      node = node.parentElement;
    }
    return false;
  }
  function applyTheme() {
    var root = document.getElementById(ROOT_ID); if (!root) return;
    if (!root.isConnected) { teardown(); return; }
    var dark = isDark(root.parentElement || document.body);
    if (dark === lastDark) return;
    lastDark = dark;
    var pal = dark ? DARK : LIGHT;
    for (var k in pal) if (pal.hasOwnProperty(k)) root.style.setProperty(k, pal[k]);
  }

  // ===================== INIT / TEARDOWN =====================
  function teardown() { if (healTimer) clearInterval(healTimer); healTimer = null; }
  function init() {
    try {
      applyTheme(); render(); fetchData();
      healTimer = setInterval(applyTheme, 1000); // Theme-Wechsel nahezu sofort (LS feuert kein Event)
    } catch (e) { console.error('[TC24-GL] Init error:', e); }
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', function () { try { init(); } catch (e) { console.error('[TC24-GL] Boot error:', e); } });
  } else {
    try { init(); } catch (e) { console.error('[TC24-GL] Boot error:', e); }
  }
})();
</script>
